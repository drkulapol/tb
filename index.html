#กลุ่มเสี่ยงเกณฑ์ในโค้ด

        
1ผู้ติดเชื้อ HIVICD10 B20–B24 หรืออยู่คลินิกที่ชื่อมีคำว่า HIV / ARV / เอดส์ / ยาต้านไวรัส ฯลฯ

        
2เบาหวานคุมไม่ได้(ICD10 E10–E11 หรือคลินิกเบาหวาน) และ HbA1c ล่าสุด ≥ 7

        
3ไตวายเรื้อรังICD10 N183–N185 หรือ eGFR ล่าสุด < 60

        
4อายุ ≥ 65 ปี + เบาหวานอายุ ณ วันทำ CXR ≥ 65 และ (ICD10 E10–E11 หรือคลินิกเบาหวาน)

        
5อายุ ≥ 65 ปี + COPDอายุ ≥ 65 และ (ICD10 J44 หรือคลินิก COPD/ปอด ที่ไม่ใช่คลินิกวัณโรค)

        
6ติดสุรา / สารเสพติดICD10 F10–F19 หรือคลินิกสุรา/ยาเสพติด (ไม่นับคลินิกเลิกบุหรี่) หรือคัดกรองว่าดื่ม

        
7อายุ ≥ 65 ปี + สูบบุหรี่อายุ ≥ 65 และคัดกรองล่าสุดว่าสูบ

        
8บุคลากรสาธารณสุขเลข CID ตรงกับตาราง doctor

      

      
รายการ CXR = non drug ที่ nhso_adp_code = 41003 และชื่อมีคำว่า CXR หรือ Chest · ค้นคลินิกจากชื่อคลินิก จึงใช้ได้ทุก รพ. โดยไม่ต้องรู้รหัส · เฉพาะ OPD สัญชาติไทย · 1 แถวต่อคนต่อปีงบประมาณ

    


    

      
โค้ดช่วยตรวจก่อนรัน (แนะนำ)

      
1) ดูว่ารายการ CXR ที่โค้ดจะนับมีอะไรบ้าง

      
SELECT icode, name, nhso_adp_code
FROM nondrugitems
WHERE trim(nhso_adp_code) = '41003'
  AND name ~* '(cxr|chest)'
ORDER BY icode;

      
2) ดูว่าคลินิกไหนถูกนับเป็นคลินิกกลุ่มเสี่ยง (ถ้าตกหล่นหรือเกิน ให้แก้คำค้นใน params ของโค้ดหลัก)

      
SELECT clinic, name,
       CASE
         WHEN name ~* '(hiv|arv|aids|\yart\y|napha|เอดส์|เอชไอวี|ยาต้านไวรัส|ภูมิคุ้มกันบกพร่อง)' THEN 'HIV'
         WHEN name ~* '(\ydm\y|diab|เบาหวาน)' THEN 'DM'
         WHEN name ~* '(copd|ปอดอุดกั้น|ถุงลมโป่งพอง|ปอด)'
              AND NOT name ~* '(วัณโรค|\ytb\y|tuberc)' THEN 'COPD'
         WHEN name ~* '(alcohol|substance|addict|methadone|สุรา|เหล้า|แอลกอฮอล์|เสพติด|บำบัดยา|เมทาโดน|ธัญญารักษ์)'
              AND NOT name ~* '(บุหรี่|smok)' THEN 'SUBST'
       END AS risk_group
FROM clinic
ORDER BY risk_group NULLS LAST, clinic;

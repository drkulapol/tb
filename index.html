#กลุ่มเสี่ยงเกณฑ์ในโค้ด

        
1ผู้ติดเชื้อ HIVICD10 B20–B24 หรืออยู่คลินิก ARV

        
2เบาหวานคุมไม่ได้(ICD10 E10–E11 หรือคลินิก DM) และ HbA1c ล่าสุด ≥ 7

        
3ไตวายเรื้อรังICD10 N183–N185 หรือ eGFR ล่าสุด < 60

        
4อายุ ≥ 65 ปี + เบาหวานอายุ ณ วันทำ CXR ≥ 65 และ (ICD10 E10–E11 หรือคลินิก DM)

        
5อายุ ≥ 65 ปี + COPDอายุ ≥ 65 และ (ICD10 J44 หรือคลินิก COPD)

        
6ติดสุรา / สารเสพติดICD10 F10–F19 หรือคลินิก Alcohol/substance หรือคัดกรองว่าดื่ม

        
7อายุ ≥ 65 ปี + สูบบุหรี่อายุ ≥ 65 และคัดกรองล่าสุดว่าสูบ

        
8บุคลากรสาธารณสุขเลข CID ตรงกับตาราง doctor

โค้ดช่วยหารหัสของโรงพยาบาลท่าน

      
หารหัส CXR ใน non drug

      
SELECT icode, name, istatus
FROM nondrugitems
WHERE name ILIKE '%chest%' OR name ILIKE '%CXR%' OR name ILIKE '%ทรวงอก%'
ORDER BY icode;

      
ดูรหัสคลินิกทั้งหมด

      
SELECT clinic, name
FROM clinic
ORDER BY clinic;

      
หารหัส Lab HbA1c / eGFR (ถ้าค้นจากชื่อแล้วได้ผลไม่ตรง)

      
SELECT lab_items_code, lab_items_name
FROM lab_items
WHERE lab_items_name ~* '(a1c|glycat|gfr)'
ORDER BY lab_items_name;
